# std/ledger.pengu - The Ledger (CSV / TSV Processing)
# Provides comprehensive CSV and TSV reading, writing, and parsing.

import std.spark
import std.oracle
import std.scrolls
import std.archivum

# ----------------------------------------------------------------------------
# External C Runtime Declarations
# ----------------------------------------------------------------------------

declare pengu_c_ledger_parse_csv with data as string and delimiter as string and has_header as bool into maybe list of list of string
declare pengu_c_ledger_generate_csv with rows as list of list of string and delimiter as string into maybe string
declare pengu_c_ledger_parse_line with line as string and delimiter as string into maybe list of string
declare pengu_c_ledger_escape_field with field as string and delimiter as string into string
declare pengu_c_ledger_read_file with path as string and delimiter as string and has_header as bool into maybe list of list of string
declare pengu_c_ledger_write_file with path as string and rows as list of list of string and delimiter as string into bool
declare pengu_c_ledger_detect_delimiter with data as string into maybe string

# ----------------------------------------------------------------------------
# CSV / TSV Functional API
# ----------------------------------------------------------------------------

weave read_csv with path as string and has_header as bool is false and delimiter as string is "," into maybe list of list of string:
    return calling pengu_c_ledger_read_file with path and delimiter and has_header

weave read_tsv with path as string and has_header as bool is false into maybe list of list of string:
    return calling pengu_c_ledger_read_file with path and "\t" and has_header

weave write_csv with path as string and rows as list of list of string and delimiter as string is "," into bool:
    return calling pengu_c_ledger_write_file with path and rows and delimiter

weave write_tsv with path as string and rows as list of list of string into bool:
    return calling pengu_c_ledger_write_file with path and rows and "\t"

weave parse_csv with data as string and delimiter as string is "," and has_header as bool is false into maybe list of list of string:
    return calling pengu_c_ledger_parse_csv with data and delimiter and has_header

weave parse_tsv with data as string and has_header as bool is false into maybe list of list of string:
    return calling pengu_c_ledger_parse_csv with data and "\t" and has_header

weave to_csv_string with rows as list of list of string and delimiter as string is "," into maybe string:
    return calling pengu_c_ledger_generate_csv with rows and delimiter

weave to_tsv_string with rows as list of list of string into maybe string:
    return calling pengu_c_ledger_generate_csv with rows and "\t"

weave parse_line with line as string and delimiter as string is "," into maybe list of string:
    return calling pengu_c_ledger_parse_line with line and delimiter

weave detect_delimiter with data as string into maybe string:
    return calling pengu_c_ledger_detect_delimiter with data

weave escape_field with field as string and delimiter as string is "," into string:
    return calling pengu_c_ledger_escape_field with field and delimiter
